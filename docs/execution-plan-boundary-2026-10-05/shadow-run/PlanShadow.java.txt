// THE SHADOW RUN's probe (docs/execution-plan-boundary-2026-10-05/shadow-run/): applied by probe.py, never landed.
package com.legend;

/**
 * Today's text answer beside the lite plan's, byte for byte: after today's path answers (a CSV or JSON wire, a stream
 * of JSON rows, execute's JSON rows), the same query is planned ({@code TypedQuery.executionPlan}) and run by the
 * runner on a session of the same source, and the two texts compared. One line per call, appended to
 * {@code $TEST_UNDECLARED_OUTPUTS_DIR/shadow-<pid>.tsv}. Today's answer is never changed: the shadow's own failures are
 * recorded, never thrown, and its statements are not counted ({@code StatementOrigin.SHADOW}).
 */
public final class PlanShadow {

    private PlanShadow() {
    }

    private static final Object LOCK = new Object();

    /** A writer that passes everything to {@code out} and keeps a copy: today's answer, for the comparison. */
    public static final class Tee extends java.io.Writer {
        private final java.io.Writer out;
        private final StringBuilder copy = new StringBuilder();

        public Tee(java.io.Writer out) {
            this.out = out;
        }

        @Override
        public void write(char[] cbuf, int off, int len) throws java.io.IOException {
            out.write(cbuf, off, len);
            copy.append(cbuf, off, len);
        }

        @Override
        public void flush() throws java.io.IOException {
            out.flush();
        }

        @Override
        public void close() throws java.io.IOException {
            out.close();
        }

        public String text() {
            return copy.toString();
        }
    }

    public static void compare(String kind, String model, Object query, @com.legend.base.Nullable String runtime,
            TypedQuery.Output output, String today, com.legend.exec.Sessions.Source sessions) {
        if (com.legend.exec.StatementOrigin.SHADOW.get()) {
            return;
        }
        com.legend.exec.StatementOrigin.SHADOW.set(true);
        try {
            com.legend.compiler.element.ModelContext ctx = Compiler.compileModel(model);
            TypedQuery q = query instanceof String s ? Compiler.query(ctx, s)
                    : Compiler.query(ctx, (com.legend.protocol.spec.ValueSpecification) query);
            var planned0 = plan(q, runtime, output);
            if (planned0 instanceof Throwable e) {
                record("PLAN_REFUSED", kind, runtime, query, e.getClass().getSimpleName() + ": " + e.getMessage());
                return;
            }
            var plan = q.executionPlan(runtime, output);
            java.io.StringWriter out = new java.io.StringWriter();
            try (com.legend.exec.Sessions.Session session = sessions.open(Compiler.executesOn(ctx, runtime), ctx)) {
                com.legend.exec.PlanRunner.run(plan, java.util.Map.of(),
                        com.legend.exec.PlanSessions.given(session.connection()), out);
            } catch (Throwable e) {
                record("RUN_FAILED", kind, runtime, query, e.getClass().getSimpleName() + ": " + e.getMessage());
                return;
            }
            String planned = out.toString();
            if (planned.equals(today)) {
                record("SAME", kind, runtime, query, "");
            } else {
                int i = 0;
                while (i < planned.length() && i < today.length() && planned.charAt(i) == today.charAt(i)) {
                    i++;
                }
                int from = Math.max(0, i - 40);
                record("DIFF", kind, runtime, query, "at " + i + ": today ..."
                        + today.substring(from, Math.min(today.length(), i + 80)) + " | plan ..."
                        + planned.substring(from, Math.min(planned.length(), i + 80)));
            }
        } catch (Throwable e) {
            record("SHADOW_ERROR", kind, runtime, query, e.getClass().getSimpleName() + ": " + e.getMessage());
        } finally {
            com.legend.exec.StatementOrigin.SHADOW.set(false);
        }
    }

    /** The plan, or why it is refused (the planner refuses by name what a plan does not do yet). */
    private static Object plan(TypedQuery q, @com.legend.base.Nullable String runtime, TypedQuery.Output output) {
        try {
            return q.executionPlan(runtime, output);
        } catch (Throwable e) {
            return e;
        }
    }

    private static void record(String verdict, String kind, @com.legend.base.Nullable String runtime, Object query,
            String detail) {
        String dir = System.getenv("TEST_UNDECLARED_OUTPUTS_DIR");
        if (dir == null) {
            return;
        }
        String line = String.join("\t", verdict, kind, String.valueOf(System.getenv("TEST_TARGET")),
                String.valueOf(runtime), oneLine(String.valueOf(query), 600), oneLine(detail, 900)) + "\n";
        synchronized (LOCK) {
            try {
                java.nio.file.Files.writeString(java.nio.file.Path.of(dir, "shadow-" + ProcessHandle.current().pid() + ".tsv"),
                        line, java.nio.file.StandardOpenOption.CREATE, java.nio.file.StandardOpenOption.APPEND);
            } catch (java.io.IOException e) {
                // the record is lost; today's answer is not touched
            }
        }
    }

    private static String oneLine(String s, int max) {
        String t = s.replace('\t', ' ').replace('\n', ' ').replace('\r', ' ');
        return t.length() > max ? t.substring(0, max) + "..." : t;
    }
}
